
import { Vec2, clamp, lerp } from './physics';

export type BlobCharacterType =
  | 'pink_dome'
  | 'yellow_star'
  | 'mint_bean'
  | 'lilac_flower'
  | 'gold_coin';

type Bounds = {
  left: number;
  right: number;
  top: number;
  bottom: number;
};

type Style = {
  light: string;
  middle: string;
  dark: string;
  rim: string;
  ink: string;
  glow: string;
  mood: number;
};

type Point = {
  x: number;
  y: number;
  vx: number;
  vy: number;
};

type Expression =
  | 'idle'
  | 'curious'
  | 'stretch'
  | 'squish'
  | 'startled';

const TAU = Math.PI * 2;
const COUNT = 20;

const COLORS: Record<BlobCharacterType, Style> = {
  pink_dome: {
    light: '#FFC3E1',
    middle: '#FF65A2',
    dark: '#CA216C',
    rim: '#8E154D',
    ink: '#49122D',
    glow: '255,188,221',
    mood: 0,
  },
  yellow_star: {
    light: '#FFF7B9',
    middle: '#FFD75D',
    dark: '#E9A42B',
    rim: '#A96D1D',
    ink: '#68401A',
    glow: '255,234,153',
    mood: 1,
  },
  mint_bean: {
    light: '#D7FFEE',
    middle: '#78E7CA',
    dark: '#26B8AD',
    rim: '#138881',
    ink: '#17504E',
    glow: '188,255,226',
    mood: 2,
  },
  lilac_flower: {
    light: '#F9E1FF',
    middle: '#D8A5F0',
    dark: '#9669CC',
    rim: '#65439B',
    ink: '#4F306F',
    glow: '235,193,255',
    mood: 3,
  },
  gold_coin: {
    light: '#FFE2EE',
    middle: '#FFA9C8',
    dark: '#DD76A4',
    rim: '#A44D7A',
    ink: '#71334F',
    glow: '255,207,227',
    mood: 4,
  },
};

function difference(a: number, b: number): number {
  return Math.atan2(Math.sin(a - b), Math.cos(a - b));
}

function smooth(
  current: number,
  target: number,
  dt: number,
  speed: number
): number {
  return lerp(
    current,
    target,
    1 - Math.exp(-speed * dt)
  );
}

export class LivingCharacter {
  type: BlobCharacterType;

  x: number;
  y: number;
  vx = 0;
  vy = 0;
  radius: number;

  normX: number;
  normY: number;

  isDragging = false;
  dragOffset: Vec2 = { x: 0, y: 0 };
  lastDragPos: Vec2 = { x: 0, y: 0 };

  wobble = 0;
  wobbleVel = 0;
  rotation = 0;
  vRot = 0;
  idlePhase = Math.random() * TAU;

  driftHeading = Math.random() * TAU;
  targetHeading = Math.random() * TAU;
  driftSpeed: number;
  turnTimer = 2 + Math.random() * 3;

  eyeOffset: Vec2 = { x: 0, y: 0 };
  targetEyeOffset: Vec2 = { x: 0, y: 0 };
  blinkTimer = 2 + Math.random() * 4;
  blinkProgress = 0;

  readonly numPoints = COUNT;

  // Keep these fields for compatibility/debugging.
  baseOffsets: number[] = [];
  pointDeltas: number[] = [];
  pointVelocities: number[] = [];

  stretch = 0;
  stretchVel = 0;
  stretchAngle = 0;

  private points: Point[] = [];
  private rest: Vec2[] = [];
  private shapeSeed = Math.random() * TAU;
  private style: Style;

  private pointer: Vec2 = { x: 0, y: 0 };
  private dragIndex = 0;
  private dragStrength = 0;

  private expression: Expression = 'idle';
  private expressionTimer = 0;
  private faceIntensity = 0;
  private faceSquash = 0;

  private gazeTimer = 0;
  private idleLook = 0;
  private collisionEnergy = 0;
  private elapsed = 0;

  private faceX = 0;
  private faceY = 0;
  private faceVX = 0;
  private faceVY = 0;

  private readonly basePointStiffness = 30;
  private readonly neighborStiffness = 100;
  private readonly radialDamping = 7;

  constructor(
    type: BlobCharacterType,
    normX: number,
    normY: number,
    radius = 54
  ) {
    this.type = type;
    this.normX = normX;
    this.normY = normY;
    this.radius = radius;

    this.x = normX * 390;
    this.y = normY * 693;

    this.style = COLORS[type];

    const speeds: Record<BlobCharacterType, number> = {
      pink_dome: 11,
      yellow_star: 17,
      mint_bean: 14,
      lilac_flower: 8,
      gold_coin: 12,
    };

    this.driftSpeed = speeds[type];
    this.initializeBody();
  }

  private initializeBody() {
    this.points = [];
    this.rest = [];
    this.baseOffsets = [];
    this.pointDeltas = [];
    this.pointVelocities = [];

    const mood = this.style.mood;

    for (let i = 0; i < COUNT; i++) {
      const a = (i / COUNT) * TAU;

      // Organic, but intentionally not a star/flower/coin.
      const contour =
        1 +
        Math.sin(a * 2 + mood * 0.7) * 0.035 +
        Math.cos(a * 3 + this.shapeSeed) * 0.02 +
        Math.sin(a) * 0.025;

      const r = this.radius * contour;

      const p = {
        x: Math.cos(a) * r,
        y: Math.sin(a) * r,
      };

      this.rest.push({ ...p });
      this.points.push({ ...p, vx: 0, vy: 0 });

      this.baseOffsets.push(r - this.radius);
      this.pointDeltas.push(0);
      this.pointVelocities.push(0);
    }
  }

  updateProportions(stageW: number, stageH: number) {
    this.x = this.normX * stageW;
    this.y = this.normY * stageH;
  }

  contains(p: Vec2): boolean {
    const dx = p.x - this.x;
    const dy = p.y - this.y;

    // Slightly generous touch area for mobile.
    return Math.hypot(dx, dy) <= this.radius * 1.15;
  }

  private setExpression(
    expression: Expression,
    seconds: number
  ) {
    this.expression = expression;
    this.expressionTimer = seconds;
  }

  private closestPoint(angle: number): number {
    let best = 0;
    let bestDistance = Infinity;

    for (let i = 0; i < COUNT; i++) {
      const a = Math.atan2(
        this.points[i].y,
        this.points[i].x
      );

      const d = Math.abs(difference(a, angle));

      if (d < bestDistance) {
        bestDistance = d;
        best = i;
      }
    }

    return best;
  }

  private deform(
    angle: number,
    amount: number,
    spread = 0.75
  ) {
    for (let i = 0; i < COUNT; i++) {
      const a = (i / COUNT) * TAU;
      const d = difference(a, angle);

      const influence = Math.exp(
        -(d * d) / (2 * spread * spread)
      );

      this.points[i].x +=
        Math.cos(a) * amount * influence;

      this.points[i].y +=
        Math.sin(a) * amount * influence;
    }
  }

  applyContact(
    normalX: number,
    normalY: number,
    penetration: number
  ) {
    const len = Math.hypot(normalX, normalY);

    if (len < 0.0001 || penetration <= 0) return;

    const angle = Math.atan2(normalY, normalX);

    // Visible surface compression, capped for stability.
    const depth = Math.min(
      penetration * 0.7,
      this.radius * 0.34
    );

    const compression = Math.max(
      0.04,
      depth / this.radius
    );

    for (let i = 0; i < COUNT; i++) {
      const a = (i / COUNT) * TAU;
      const d = Math.abs(difference(a, angle));

      const dent = Math.exp(
        -(d * d) / (2 * 0.48 * 0.48)
      );

      const shoulder = d - 1.0;
      const bulge = Math.exp(
        -(shoulder * shoulder) /
        (2 * 0.32 * 0.32)
      );

      const displacement =
        -depth * dent + depth * 0.35 * bulge;

      const p = this.points[i];

      // Shape changes at the contact patch rather than
      // applying a large body-wide launch impulse.
      const targetX =
        this.rest[i].x +
        Math.cos(a) * displacement;

      const targetY =
        this.rest[i].y +
        Math.sin(a) * displacement;

      const blend = Math.min(0.55, compression);

      p.x = lerp(p.x, targetX, blend);
      p.y = lerp(p.y, targetY, blend);

      p.vx *= 0.8;
      p.vy *= 0.8;
    }

    this.collisionEnergy = Math.max(
      this.collisionEnergy,
      compression
    );

    this.faceSquash = Math.max(
      this.faceSquash,
      compression
    );

    this.setExpression('squish', 0.34);
  }

  startDrag(p: Vec2) {
    this.isDragging = true;

    this.dragOffset = {
      x: this.x - p.x,
      y: this.y - p.y,
    };

    this.lastDragPos = { ...p };
    this.pointer = { ...p };

    this.vx = 0;
    this.vy = 0;

    const angle = Math.atan2(
      p.y - this.y,
      p.x - this.x
    );

    this.dragIndex = this.closestPoint(angle);
    this.dragStrength = 1;

    this.deform(angle, -this.radius * 0.10);

    this.setExpression('curious', 0.5);
    this.faceIntensity = 0.6;
  }

  drag(p: Vec2, dt: number) {
    if (!this.isDragging) return;

    const safeDt = Math.max(dt, 1 / 120);

    const deltaX = p.x - this.lastDragPos.x;
    const deltaY = p.y - this.lastDragPos.y;

    const pointerVX = clamp(
      deltaX / safeDt,
      -650,
      650
    );

    const pointerVY = clamp(
      deltaY / safeDt,
      -650,
      650
    );

    this.vx = lerp(this.vx, pointerVX * 0.28, 0.2);
    this.vy = lerp(this.vy, pointerVY * 0.28, 0.2);

    this.pointer = { ...p };
    this.lastDragPos = { ...p };

    // The core follows the pointer with a visible delay.
    const targetX = p.x + this.dragOffset.x;
    const targetY = p.y + this.dragOffset.y;

    const follow = 1 - Math.exp(-safeDt * 6);

    this.x = lerp(this.x, targetX, follow);
    this.y = lerp(this.y, targetY, follow);

    const speed = Math.hypot(pointerVX, pointerVY);

    if (speed > 45) {
      this.stretchAngle = Math.atan2(
        pointerVY,
        pointerVX
      );

      this.stretch = smooth(
        this.stretch,
        clamp(speed / 1100, 0, 0.32),
        safeDt,
        8
      );

      this.setExpression('stretch', 0.25);
      this.faceIntensity = smooth(
        this.faceIntensity,
        1,
        safeDt,
        12
      );
    }

    this.targetEyeOffset = {
      x: clamp((p.x - this.x) * 0.14, -13, 13),
      y: clamp((p.y - this.y) * 0.14, -11, 11),
    };
  }

  endDrag() {
    this.isDragging = false;
    this.dragStrength = 0;

    // Heavy jelly: preserve some movement, not full flick speed.
    this.vx *= 0.25;
    this.vy *= 0.25;

    // Release stored tension at the dragged surface.
    const p = this.points[this.dragIndex];

    p.vx *= 0.8;
    p.vy *= 0.8;

    this.stretchVel = -1.1;
    this.wobble = 0.16;
    this.wobbleVel = -2;

    this.setExpression('startled', 0.32);
  }

  tap() {
    const angle = Math.random() * TAU;

    this.deform(angle, -this.radius * 0.16);

    this.wobble = 0.2;
    this.wobbleVel = -3;

    this.faceIntensity = 0.9;
    this.setExpression('startled', 0.38);
  }

  private updateSurface(dt: number) {
    // Semi-implicit integration in several stable substeps.
    const steps = Math.max(1, Math.ceil(dt / (1 / 120)));
    const h = dt / steps;

    for (let substep = 0; substep < steps; substep++) {
      const forces: Vec2[] = Array.from(
        { length: COUNT },
        () => ({ x: 0, y: 0 })
      );

      // Core-to-skin springs.
      for (let i = 0; i < COUNT; i++) {
        const p = this.points[i];
        const rest = this.rest[i];

        const idleWave =
          Math.sin(
            this.elapsed * 1.6 +
            (i / COUNT) * TAU * 2 +
            this.shapeSeed
          ) * this.radius * 0.018;

        const angle = (i / COUNT) * TAU;

        const targetX =
          rest.x + Math.cos(angle) * idleWave;

        const targetY =
          rest.y + Math.sin(angle) * idleWave;

        forces[i].x +=
          (targetX - p.x) * this.basePointStiffness;

        forces[i].y +=
          (targetY - p.y) * this.basePointStiffness;
      }

      // Neighbour springs: preserve soft perimeter continuity.
      for (let i = 0; i < COUNT; i++) {
        const j = (i + 1) % COUNT;

        const a = this.points[i];
        const b = this.points[j];

        const ra = this.rest[i];
        const rb = this.rest[j];

        const restLength = Math.hypot(
          rb.x - ra.x,
          rb.y - ra.y
        );

        const dx = b.x - a.x;
        const dy = b.y - a.y;
        const len = Math.max(
          0.001,
          Math.hypot(dx, dy)
        );

        const stretch = len - restLength;

        const force =
          stretch * this.neighborStiffness;

        const fx = (dx / len) * force;
        const fy = (dy / len) * force;

        forces[i].x += fx;
        forces[i].y += fy;

        forces[j].x -= fx;
        forces[j].y -= fy;
      }

      // Area preservation: outward pressure resists collapse.
      let signedArea = 0;

      for (let i = 0; i < COUNT; i++) {
        const a = this.points[i];
        const b = this.points[(i + 1) % COUNT];

        signedArea += a.x * b.y - b.x * a.y;
      }

      const area = Math.abs(signedArea) * 0.5;
      const targetArea = Math.PI * this.radius * this.radius;

      const pressure = clamp(
        (targetArea - area) / targetArea,
        -0.35,
        0.35
      );

      for (let i = 0; i < COUNT; i++) {
        const p = this.points[i];

        const length = Math.max(
          1,
          Math.hypot(p.x, p.y)
        );

        forces[i].x +=
          (p.x / length) * pressure * 130;

        forces[i].y +=
          (p.y / length) * pressure * 130;
      }

      // A dragged point is controlled by a strong but
      // compliant spring; the rest of the surface follows.
      if (this.isDragging) {
        const idx = this.dragIndex;
        const point = this.points[idx];

        // Drag target in the blob's local coordinates.
        const targetX = this.pointer.x - this.x;
        const targetY = this.pointer.y - this.y;

        const maxDist = this.radius * 2.0;
        const length = Math.hypot(targetX, targetY);
        const factor =
          length > maxDist ? maxDist / length : 1;

        forces[idx].x +=
          (targetX * factor - point.x) * 360;

        forces[idx].y +=
          (targetY * factor - point.y) * 360;
      }

      const damping = Math.exp(
        -this.radialDamping * h
      );

      for (let i = 0; i < COUNT; i++) {
        const p = this.points[i];

        p.vx += forces[i].x * h;
        p.vy += forces[i].y * h;

        p.vx *= damping;
        p.vy *= damping;

        // Prevent pathological spring explosions.
        p.vx = clamp(p.vx, -450, 450);
        p.vy = clamp(p.vy, -450, 450);

        p.x += p.vx * h;
        p.y += p.vy * h;

        const maxR = this.radius * 2.2;
        const r = Math.hypot(p.x, p.y);

        if (r > maxR) {
          p.x *= maxR / r;
          p.y *= maxR / r;
          p.vx *= 0.6;
          p.vy *= 0.6;
        }
      }
    }

    for (let i = 0; i < COUNT; i++) {
      const p = this.points[i];
      const a = (i / COUNT) * TAU;

      this.pointDeltas[i] =
        Math.hypot(p.x, p.y) -
        Math.hypot(this.rest[i].x, this.rest[i].y);

      this.pointVelocities[i] =
        p.vx * Math.cos(a) +
        p.vy * Math.sin(a);
    }
  }

  private updateFace(dt: number) {
    this.expressionTimer -= dt;

    if (this.expressionTimer <= 0 && !this.isDragging) {
      this.expression = 'idle';
    }

    this.faceIntensity = smooth(
      this.faceIntensity,
      this.isDragging ? 0.85 : 0,
      dt,
      this.isDragging ? 7 : 3
    );

    this.faceSquash = smooth(
      this.faceSquash,
      0,
      dt,
      5
    );

    this.collisionEnergy = smooth(
      this.collisionEnergy,
      0,
      dt,
      4
    );

    // The face itself has inertia, separate from the core.
    const faceTargetX = clamp(
      -this.vx * 0.035,
      -11,
      11
    );

    const faceTargetY = clamp(
      -this.vy * 0.035,
      -9,
      9
    );

    this.faceVX +=
      (faceTargetX - this.faceX) * 90 * dt -
      this.faceVX * 12 * dt;

    this.faceVY +=
      (faceTargetY - this.faceY) * 90 * dt -
      this.faceVY * 12 * dt;

    this.faceX += this.faceVX * dt;
    this.faceY += this.faceVY * dt;

    // Different personalities look around at different rates.
    this.gazeTimer -= dt;

    if (this.gazeTimer <= 0 && !this.isDragging) {
      const mood = this.style.mood;

      this.idleLook = (Math.random() - 0.5) * 2;

      this.gazeTimer =
        mood === 3
          ? 1.2 + Math.random() * 1.3
          : 2 + Math.random() * 3;

      if (mood === 2) {
        this.setExpression('curious', 0.6);
      }
    }

    if (!this.isDragging) {
      this.targetEyeOffset = {
        x: this.idleLook * 5,
        y: Math.sin(this.elapsed * 0.8) * 2,
      };
    }

    this.eyeOffset.x = smooth(
      this.eyeOffset.x,
      this.targetEyeOffset.x,
      dt,
      8
    );

    this.eyeOffset.y = smooth(
      this.eyeOffset.y,
      this.targetEyeOffset.y,
      dt,
      8
    );

    this.blinkTimer -= dt;

    if (this.blinkTimer <= 0) {
      this.blinkProgress += dt * 10;

      if (this.blinkProgress >= Math.PI) {
        this.blinkProgress = 0;

        this.blinkTimer =
          this.style.mood === 3
            ? 1.7 + Math.random() * 2
            : 2.4 + Math.random() * 3.5;
      }
    }
  }

  update(
    dt: number,
    bounds: Bounds,
    stageW: number,
    stageH: number,
    time: number
  ) {
    // Physics and visuals use bounded frame delta.
    const safeDt = clamp(dt, 0, 0.04);

    this.elapsed += safeDt;
    this.idlePhase += safeDt;

    if (!this.isDragging) {
      this.turnTimer -= safeDt;

      if (this.turnTimer <= 0) {
        this.turnTimer = 2.5 + Math.random() * 3;
        this.targetHeading = Math.random() * TAU;
      }

      this.driftHeading += difference(
        this.targetHeading,
        this.driftHeading
      ) * Math.min(1, safeDt);

      const targetVX =
        Math.cos(this.driftHeading) *
        this.driftSpeed;

      const targetVY =
        Math.sin(this.driftHeading) *
        this.driftSpeed;

      this.vx = smooth(
        this.vx,
        targetVX,
        safeDt,
        1.4
      );

      this.vy = smooth(
        this.vy,
        targetVY,
        safeDt,
        1.4
      );

      this.x += this.vx * safeDt;
      this.y += this.vy * safeDt;
    }

    // Do not let whole-body collision send characters flying.
    const maxVelocity = 230;
    const speed = Math.hypot(this.vx, this.vy);

    if (speed > maxVelocity) {
      this.vx = this.vx / speed * maxVelocity;
      this.vy = this.vy / speed * maxVelocity;
    }

    const r = this.radius;

    if (this.x < bounds.left + r) {
      this.x = bounds.left + r;
      this.vx = Math.max(0, this.vx * -0.2);

      this.applyContact(-1, 0, r * 0.13);
    }

    if (this.x > bounds.right - r) {
      this.x = bounds.right - r;
      this.vx = Math.min(0, this.vx * -0.2);

      this.applyContact(1, 0, r * 0.13);
    }

    if (this.y < bounds.top + r) {
      this.y = bounds.top + r;
      this.vy = Math.max(0, this.vy * -0.2);

      this.applyContact(0, -1, r * 0.13);
    }

    if (this.y > bounds.bottom - r) {
      this.y = bounds.bottom - r;
      this.vy = Math.min(0, this.vy * -0.2);

      this.applyContact(0, 1, r * 0.13);
    }

    this.rotation = smooth(
      this.rotation,
      Math.sin(this.elapsed * 0.9) * 0.035,
      safeDt,
      2
    );

    this.vRot *= Math.exp(-safeDt * 5);

    this.stretchVel += (
      -45 * this.stretch -
      10 * this.stretchVel
    ) * safeDt;

    this.stretch += this.stretchVel * safeDt;

    this.stretch = clamp(
      this.stretch,
      -0.15,
      0.36
    );

    this.wobbleVel += (
      -75 * this.wobble -
      12 * this.wobbleVel
    ) * safeDt;

    this.wobble += this.wobbleVel * safeDt;

    this.updateSurface(safeDt);
    this.updateFace(safeDt);

    if (stageW > 0 && stageH > 0) {
      this.normX = this.x / stageW;
      this.normY = this.y / stageH;
    }

    void time;
  }

  private bodyPath(ctx: CanvasRenderingContext2D) {
    const pts = this.points;

    const first = pts[0];
    const last = pts[COUNT - 1];

    ctx.beginPath();

    ctx.moveTo(
      (first.x + last.x) * 0.5,
      (first.y + last.y) * 0.5
    );

    for (let i = 0; i < COUNT; i++) {
      const a = pts[i];
      const b = pts[(i + 1) % COUNT];

      ctx.quadraticCurveTo(
        a.x,
        a.y,
        (a.x + b.x) * 0.5,
        (a.y + b.y) * 0.5
      );
    }

    ctx.closePath();
  }

  private drawFace(ctx: CanvasRenderingContext2D) {
    const r = this.radius;
    const ink = this.style.ink;

    // Whole face reacts to surface deformation.
    const left = this.points[Math.round(COUNT / 2)];
    const right = this.points[0];
    const top = this.points[Math.round(COUNT * 0.75)];
    const bottom = this.points[Math.round(COUNT * 0.25)];

    const horizontalSize = Math.max(
      r * 0.9,
      right.x - left.x
    );

    const verticalSize = Math.max(
      r * 0.9,
      bottom.y - top.y
    );

    const squashX = clamp(
      horizontalSize / (r * 2),
      0.65,
      1.4
    );

    const squashY = clamp(
      verticalSize / (r * 2),
      0.65,
      1.4
    );

    const expression = this.expression;

    ctx.save();

    ctx.translate(
      this.faceX,
      this.faceY
    );

    ctx.scale(
      lerp(1, squashX, 0.55),
      lerp(1, squashY, 0.55)
    );

    const blink = Math.sin(this.blinkProgress);
    const blinkScale = 1 - blink * 0.94;

    const surprise =
      expression === 'startled' ||
      expression === 'curious';

    const stretching =
      expression === 'stretch';

    const squishing =
      expression === 'squish';

    const eyeOpen =
      surprise ? 1.3 :
        stretching ? 0.85 :
          squishing ? 0.28 : 1;

    const eyeHeight = Math.max(
      0.7,
      7.6 * eyeOpen * blinkScale
    );

    const eyeWidth =
      surprise ? 5.6 : squishing ? 6 : 4.5;

    const eyeSpacing =
      r * (squishing ? 0.22 : 0.25);

    const gazeX = this.eyeOffset.x;
    const gazeY = this.eyeOffset.y;

    // Blush.
    if (this.style.mood !== 3) {
      ctx.fillStyle = 'rgba(255,95,139,0.18)';

      for (const side of [-1, 1]) {
        ctx.beginPath();

        ctx.ellipse(
          side * r * 0.40,
          r * 0.11,
          r * 0.11,
          r * 0.058,
          0,
          0,
          TAU
        );

        ctx.fill();
      }
    }

    // Eyes.
    ctx.fillStyle = ink;

    for (const side of [-1, 1]) {
      const x = side * eyeSpacing + gazeX;
      const y = -r * 0.07 + gazeY;

      ctx.beginPath();

      ctx.ellipse(
        x,
        y,
        eyeWidth,
        eyeHeight,
        0,
        0,
        TAU
      );

      ctx.fill();

      if (eyeHeight > 3) {
        ctx.fillStyle = 'rgba(255,255,255,0.92)';

        ctx.beginPath();

        ctx.arc(
          x - 1.2,
          y - eyeHeight * 0.3,
          1.65,
          0,
          TAU
        );

        ctx.fill();
        ctx.fillStyle = ink;
      }
    }

    // Mouth changes shape according to the active state.
    const mouthY = r * 0.16;

    ctx.strokeStyle = ink;
    ctx.fillStyle = ink;
    ctx.lineWidth = 2.3;
    ctx.lineCap = 'round';

    if (expression === 'startled' ||
      expression === 'stretch') {
      ctx.beginPath();

      ctx.ellipse(
        0,
        mouthY,
        stretching ? 4.8 : 5.2,
        stretching ? 7 : 6,
        0,
        0,
        TAU
      );

      ctx.fill();
    } else if (squishing) {
      ctx.beginPath();

      ctx.moveTo(-6, mouthY);
      ctx.quadraticCurveTo(
        0,
        mouthY + 2,
        6,
        mouthY
      );

      ctx.stroke();
    } else if (expression === 'curious') {
      ctx.beginPath();

      ctx.ellipse(
        0,
        mouthY,
        3.3,
        4.4,
        0,
        0,
        TAU
      );

      ctx.fill();
    } else {
      const smile =
        this.style.mood === 1 ||
          this.style.mood === 2
          ? 9
          : 5.5;

      ctx.beginPath();

      ctx.moveTo(-smile, mouthY - 2);

      ctx.quadraticCurveTo(
        0,
        mouthY + smile * 0.8,
        smile,
        mouthY - 2
      );

      ctx.stroke();
    }

    ctx.restore();
  }

  draw(ctx: CanvasRenderingContext2D, time: number) {
    const r = this.radius;
    const s = this.style;

    ctx.save();

    ctx.translate(this.x, this.y);
    ctx.rotate(this.rotation);

    // Local deformation is the main source of elasticity.
    // Keep the global transform deliberately subtle.
    const stretchScale = 1 + this.stretch * 0.15;

    ctx.rotate(this.stretchAngle);
    ctx.scale(stretchScale, 1 / stretchScale);
    ctx.rotate(-this.stretchAngle);

    // Organic soft shadow.
    ctx.save();

    ctx.shadowColor = 'rgba(12,2,26,0.45)';
    ctx.shadowBlur = 20;
    ctx.shadowOffsetY = 10;

    this.bodyPath(ctx);

    ctx.fillStyle = s.dark;
    ctx.fill();

    ctx.restore();

    const body = ctx.createRadialGradient(
      -r * 0.34,
      -r * 0.40,
      r * 0.02,
      r * 0.16,
      r * 0.18,
      r * 1.35
    );

    body.addColorStop(0, s.light);
    body.addColorStop(0.34, s.middle);
    body.addColorStop(0.8, s.dark);
    body.addColorStop(1, s.rim);

    this.bodyPath(ctx);

    ctx.fillStyle = body;
    ctx.fill();

    // All specular effects are clipped to the moving skin.
    ctx.save();

    this.bodyPath(ctx);
    ctx.clip();

    const bloom = ctx.createRadialGradient(
      -r * 0.38,
      -r * 0.45,
      0,
      -r * 0.38,
      -r * 0.45,
      r * 0.8
    );

    bloom.addColorStop(
      0,
      'rgba(255,255,255,0.54)'
    );

    bloom.addColorStop(
      0.5,
      'rgba(255,255,255,0.16)'
    );

    bloom.addColorStop(
      1,
      'rgba(255,255,255,0)'
    );

    ctx.fillStyle = bloom;

    ctx.fillRect(
      -r * 1.5,
      -r * 1.5,
      r * 3,
      r * 3
    );

    const subsurface = ctx.createRadialGradient(
      r * 0.1,
      r * 0.55,
      0,
      r * 0.1,
      r * 0.55,
      r
    );

    subsurface.addColorStop(
      0,
      `rgba(${s.glow},0.34)`
    );

    subsurface.addColorStop(
      1,
      `rgba(${s.glow},0)`
    );

    ctx.fillStyle = subsurface;

    ctx.fillRect(
      -r * 1.5,
      -r * 1.5,
      r * 3,
      r * 3
    );

    // Gently moving highlight.
    const glintX =
      -r * 0.35 +
      Math.sin(time * 0.001) * r * 0.04;

    const glint = ctx.createRadialGradient(
      glintX,
      -r * 0.48,
      0,
      glintX,
      -r * 0.48,
      r * 0.19
    );

    glint.addColorStop(
      0,
      'rgba(255,255,255,0.36)'
    );

    glint.addColorStop(
      1,
      'rgba(255,255,255,0)'
    );

    ctx.fillStyle = glint;

    ctx.fillRect(
      -r,
      -r,
      r * 2,
      r * 2
    );

    this.drawFace(ctx);

    ctx.restore();
    ctx.restore();
  }
}
